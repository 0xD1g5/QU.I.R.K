```python
def _prompt_bool(text: str, default: bool) -> bool:
    d = "Y" if default else "N"
    raw = _prompt(f"{text} (y/n)", d).lower()
    if raw in ("y", "yes"):
        return True
    if raw in ("n", "no"):
        return False
    return default
```
